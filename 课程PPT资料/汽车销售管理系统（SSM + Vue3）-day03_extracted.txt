        if (brandId != null) {
            criteria.andBrandIdEqualTo(brandId);
        }
        if (minPrice != null) {
            criteria.andPriceGreaterThanOrEqualTo(minPrice);
        }
        if (maxPrice != null) {
            criteria.andPriceLessThanOrEqualTo(maxPrice);
        }
        // 3. 执行查询（按创建时间倒序，保持列表一致性）
        example.setOrderByClause("id DESC");
        List<Car> list = carMapper.selectByExample(example);
        // 4. 返回分页对象
        return new PageInfo<>(list);
    }

            throw new BusinessException(" 价格必须大于 0");

        // 1. 校验 ID
        if (car.getId() == null) {
            throw new BusinessException(" 车辆 ID 不能为空 ");
        }
        // 2. 检查车辆是否存在
        Car existCar = carMapper.selectByPrimaryKey(car.getId());
        if (existCar == null) {
            throw new BusinessException(" 车辆不存在 ");
        }
        // 3. 判断品牌、车型、配置是否真正发生变化
        boolean brandChanged = car.getBrandId() != null && 
!car.getBrandId().equals(existCar.getBrandId());
        boolean modelChanged = car.getModel() != null && 
!car.getModel().trim().equals(existCar.getModel());
        boolean configChanged = car.getConfig() != null && 
!car.getConfig().trim().equals(existCar.getConfig());
        // 4. 只有品牌、车型、配置发生变化时才校验重复
        if (brandChanged || modelChanged || configChanged) {
            Integer finalBrandId = car.getBrandId() != null ? car.getBrandId() : 
existCar.getBrandId();
            String finalModel = car.getModel() != null ? car.getModel().trim() : 
existCar.getModel();
            String finalConfig = car.getConfig() != null ? 
car.getConfig().trim() : existCar.getConfig();
            CarExample example = new CarExample();
            example.createCriteria()
                    .andBrandIdEqualTo(finalBrandId)
                    .andModelEqualTo(finalModel)
                    .andConfigEqualTo(finalConfig)
                    .andIdNotEqualTo(car.getId());  // 排除自己
            List<Car> existList = carMapper.selectByExample(example);
            if (!existList.isEmpty()) {
                throw new BusinessException(" 该品牌下已存在相同车型和配置的车辆 ");
            }
        }
        // 5. 更新
        carMapper.updateByPrimaryKeySelective(car);
    }
