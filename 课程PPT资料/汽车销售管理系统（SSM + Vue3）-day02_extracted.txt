        // 1. 启动分页
        PageHelper.startPage(pageNum, pageSize);
        // 2. 构建查询条件（使用  MBG 生成的  Example ）
        CarExample example = new CarExample();
        CarExample.Criteria criteria = example.createCriteria();
        if (brandId != null) {
            criteria.andBrandIdEqualTo(brandId);

            criteria.andPriceGreaterThanOrEqualTo(minPrice);
        }
        if (maxPrice != null) {
            criteria.andPriceLessThanOrEqualTo(maxPrice);
        }
        // 3. 执行查询（按创建时间倒序，保持列表一致性）
        example.setOrderByClause("id DESC");
        List<Car> list = carMapper.selectByExample(example);
        // 4. 返回分页对象
        return new PageInfo<>(list);
    }
}
